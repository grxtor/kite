# Kite Plugin SDK: headers and CMake config for .mmco module development. Provides mmco_c_sdk.h and
# mmco_cxx_sdk.hpp, PluginAPI.h / PluginHooks.h / MMCOFormat.h, the transitive Kite headers they
# need, and a CMake package so out-of-tree plugins can find_package(Kite_SDK REQUIRED) and link
# Kite::SDK.

# INTERFACE library: no compiled code, only headers and include paths
add_library(Kite_sdk INTERFACE)
add_library(Kite::SDK ALIAS Kite_sdk)
set_target_properties(Kite_sdk PROPERTIES EXPORT_NAME SDK)

# C ABI INTERFACE library: Qt-free and header-only (mmco_c_sdk.h and the definitions it builds on),
# so plain C plugins, or any language that speaks the C ABI, can build a .mmco without Qt. C
# plugins link Kite::SDK_C, C++ plugins Kite::SDK.
add_library(Kite_sdk_c INTERFACE)
add_library(Kite::SDK_C ALIAS Kite_sdk_c)
set_target_properties(Kite_sdk_c PROPERTIES EXPORT_NAME SDK_C)

target_include_directories(Kite_sdk_c INTERFACE
    $<BUILD_INTERFACE:${CMAKE_SOURCE_DIR}/launcher>
    $<BUILD_INTERFACE:${CMAKE_SOURCE_DIR}/launcher/plugin/sdk>
    $<INSTALL_INTERFACE:include/kite-sdk>
    $<INSTALL_INTERFACE:include/kite-sdk/plugin/sdk>
)

# The C++ SDK inherits the C ABI include paths and adds Qt.
target_link_libraries(Kite_sdk INTERFACE Kite_sdk_c)

# Resolve include directories from imported targets (these are installed externally)
get_target_property(_KATABASIS_INCLUDE_DIRS Katabasis INTERFACE_INCLUDE_DIRECTORIES)

target_include_directories(Kite_sdk INTERFACE
    # Build-tree consumers (in-tree plugins)
    $<BUILD_INTERFACE:${CMAKE_SOURCE_DIR}/launcher>
    $<BUILD_INTERFACE:${CMAKE_SOURCE_DIR}/launcher/plugin/sdk>
    $<BUILD_INTERFACE:${CMAKE_SOURCE_DIR}/buildconfig>
    # Install-tree consumers (out-of-tree plugins)
    $<INSTALL_INTERFACE:include/kite-sdk>
    $<INSTALL_INTERFACE:include/kite-sdk/plugin/sdk>
)

target_link_libraries(Kite_sdk INTERFACE
    Qt${QT_VERSION_MAJOR}::Core
    Qt${QT_VERSION_MAJOR}::Gui
    Qt${QT_VERSION_MAJOR}::Network
)
